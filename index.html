async function shortenURL() { const longUrlInput = document.getElementById('longUrl').value; const resultBox = document.getElementById('resultBox'); const shortenedLink = document.getElementById('shortenedLink'); if (!longUrlInput) { alert('দয়া করে একটি সঠিক URL দিন!'); return; } try { const apiUrl = `${window.location.origin}/api/index`; const response = await fetch(apiUrl, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ longUrl: longUrlInput }) }); if (!response.ok) { throw new Error('সার্ভারে সমস্যা হয়েছে'); } const data = await response.json(); // এখানে আপনার চাওয়া ফরম্যাট (?q=আইডি) অনুযায়ী শর্ট লিংক তৈরি করা হচ্ছে const shortUrl = `${window.location.origin}/?q=${data.shortId}`; shortenedLink.href = shortUrl; shortenedLink.innerText = shortUrl; resultBox.style.display = 'flex'; } catch (error) { alert('দুঃখিত! লিংকটি ছোট করা যায়নি।'); console.error(error); } }
